Comprehensions e funções

numeros = [1, 2, 3, 4, 5]
quadrados = [n ** 2 for n in numeros]
pares = [n for n in numeros if n % 2 == 0]

def classificar_receita(valor, limite_alto=10000, limite_medio=5000):
    if valor > limite_alto:
        return "Alta"
    elif valor > limite_medio:
        return "Média"
    return "Baixa"

desconto = lambda preco, pct: preco * (1 - pct)

Pipeline com method chaining

resultado = (
    df_raw
    .fillna({"avaliacao": df_raw["avaliacao"].mean()})
    .assign(valor_estoque=lambda d: d["preco"] * d["estoque"])
    .query("avaliacao >= 4.0")
    [["produto", "categoria", "preco", "valor_estoque"]]
    .sort_values("valor_estoque", ascending=False)
    .reset_index(drop=True)
)

Essa mesma forma de escrever é usada no PySpark.
